cmake_minimum_required(VERSION 3.20)
project(FacetGlass VERSION 0.1.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

option(FG_BUILD_TESTS "Build core equivalence tests" ON)
option(FG_BUILD_PLUGIN "Build the OpenFX plugin (src/plugin)" ON)

set(OFX_ROOT ${CMAKE_CURRENT_SOURCE_DIR}/third_party/openfx)

# ---------------------------------------------------------------- tests
if(FG_BUILD_TESTS)
  # Reference: the Fuse v1 kernel compiled for the CPU (ground truth renders).
  add_executable(fg_fuse_harness reference/kernel-harness/harness.c)
  if(NOT MSVC)
    target_link_libraries(fg_fuse_harness m)
  endif()
  # Portable core (src/core) on the CPU backend; must match the reference.
  add_executable(fg_core_equiv tests/core_equiv.cpp)
endif()

# ---------------------------------------------------------------- 3D core (host library)
add_library(fg3d_host STATIC src/host/fg3d_scene.cpp)
target_include_directories(fg3d_host PUBLIC src/core src/host)
set_target_properties(fg3d_host PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(MSVC)
  set_target_properties(fg3d_host PROPERTIES MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_compile_options(fg3d_host PRIVATE /fp:precise)
endif()
# CUDA backend (optional: needs the CUDA Toolkit).
option(FG_ENABLE_CUDA "Build the CUDA backend if a CUDA compiler is found" ON)
set(FG_HAVE_CUDA OFF)
if(FG_ENABLE_CUDA AND NOT APPLE)
  include(CheckLanguage)
  check_language(CUDA)
  if(CMAKE_CUDA_COMPILER)
    # Turing (2080 Ti) and newer; the last entry also embeds PTX for future GPUs.
    if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
      set(CMAKE_CUDA_ARCHITECTURES 75-real 86-real 89-real 120)
    endif()
    enable_language(CUDA)
    set(FG_HAVE_CUDA ON)
    add_library(fg3d_cuda STATIC src/host/fg3d_cuda.cu)
    target_link_libraries(fg3d_cuda PUBLIC fg3d_host)
    # Exact maths (no fused multiply-add) keeps the GPU close to the CPU reference.
    target_compile_options(fg3d_cuda PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false>)
    set_target_properties(fg3d_cuda PROPERTIES CUDA_RUNTIME_LIBRARY Static POSITION_INDEPENDENT_CODE ON)
    if(MSVC)
      set_target_properties(fg3d_cuda PROPERTIES MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    endif()
    message(STATUS "FacetGlass: CUDA backend enabled (${CMAKE_CUDA_COMPILER_VERSION})")
  endif()
endif()

if(FG_BUILD_TESTS)
  if(FG_HAVE_CUDA)
    add_executable(fg3d_gpu_test tests/fg3d_gpu_test.cpp)
    target_link_libraries(fg3d_gpu_test PRIVATE fg3d_cuda)
    target_include_directories(fg3d_gpu_test PRIVATE ${CMAKE_CUDA_TOOLKIT_INCLUDE_DIRECTORIES})
    set_target_properties(fg3d_gpu_test PROPERTIES CUDA_RUNTIME_LIBRARY Static)
    find_package(CUDAToolkit REQUIRED)
    target_link_libraries(fg3d_gpu_test PRIVATE CUDA::cudart_static)
    if(MSVC)
      set_target_properties(fg3d_gpu_test PROPERTIES MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
      target_compile_options(fg3d_gpu_test PRIVATE /fp:precise)
    endif()
  endif()
  add_executable(fg3d_math_test tests/fg3d_math_test.cpp)
  target_include_directories(fg3d_math_test PRIVATE src/core)
  add_executable(fg3d_render tests/fg3d_render.cpp)
  target_link_libraries(fg3d_render PRIVATE fg3d_host)
  if(MSVC)
    set_target_properties(fg3d_render PROPERTIES MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    target_compile_options(fg3d_render PRIVATE /fp:precise)
  endif()
endif()

option(FG_BUILD_OFXHOST "Build the OFX test host (tests/ofxhost, needs expat via FetchContent)" ON)
if(FG_BUILD_TESTS AND FG_BUILD_OFXHOST)
  include(FetchContent)
  set(EXPAT_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
  set(EXPAT_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(EXPAT_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(EXPAT_BUILD_DOCS OFF CACHE BOOL "" FORCE)
  set(EXPAT_BUILD_FUZZERS OFF CACHE BOOL "" FORCE)
  set(EXPAT_BUILD_PKGCONFIG OFF CACHE BOOL "" FORCE)
  set(EXPAT_SHARED_LIBS OFF CACHE BOOL "" FORCE)
  set(EXPAT_ENABLE_INSTALL OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(expat
    URL https://github.com/libexpat/libexpat/releases/download/R_2_6_4/expat-2.6.4.tar.xz
    URL_HASH SHA256=a695629dae047055b37d50a0ff4776d1d45d0a4c842cf4ccee158441f55ff7ee
    DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_MakeAvailable(expat)

  file(GLOB OFX_HOSTSUPPORT_SRC ${OFX_ROOT}/HostSupport/src/*.cpp)
  add_executable(fg_ofxhost tests/ofxhost/fg_ofxhost.cpp ${OFX_HOSTSUPPORT_SRC})
  target_include_directories(fg_ofxhost PRIVATE ${OFX_ROOT}/include ${OFX_ROOT}/HostSupport/include)
  target_compile_definitions(fg_ofxhost PRIVATE OFX_SUPPORTS_MULTITHREAD XML_STATIC)
  target_link_libraries(fg_ofxhost PRIVATE expat)
  if(MSVC)
    target_compile_definitions(fg_ofxhost PRIVATE _CRT_SECURE_NO_WARNINGS WINDOWS)
  elseif(UNIX AND NOT APPLE)
    target_link_libraries(fg_ofxhost PRIVATE dl pthread)
  endif()
endif()

# ---------------------------------------------------------------- plugin
# TODO (next step, see CLAUDE.md): add src/plugin/*.cpp + the OFX Support
# library as a MODULE library named FacetGlass, output suffix ".ofx",
# placed in FacetGlass.ofx.bundle/Contents/<Win64|MacOS|Linux-x86-64>/.
if(FG_BUILD_PLUGIN AND EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/src/plugin/FacetGlassPlugin.cpp)
  file(GLOB OFX_SUPPORT_SRC ${OFX_ROOT}/Support/Library/*.cpp)
  file(GLOB FG_PLUGIN_SRC ${CMAKE_CURRENT_SOURCE_DIR}/src/plugin/*.cpp)
  add_library(FacetGlass MODULE ${FG_PLUGIN_SRC} ${OFX_SUPPORT_SRC})
  target_include_directories(FacetGlass PRIVATE
    ${OFX_ROOT}/include ${OFX_ROOT}/Support/include ${OFX_ROOT}/Support/Library src/core src/host)
  target_link_libraries(FacetGlass PRIVATE fg3d_host)
  if(FG_HAVE_CUDA)
    find_package(CUDAToolkit REQUIRED)
    target_link_libraries(FacetGlass PRIVATE fg3d_cuda CUDA::cudart_static)
    target_compile_definitions(FacetGlass PRIVATE FG_HAVE_CUDA)
  endif()
  set_target_properties(FacetGlass PROPERTIES PREFIX "" SUFFIX ".ofx")
  if(WIN32)
    set(FG_ARCH Win64)
  elseif(APPLE)
    set(FG_ARCH MacOS)
  else()
    set(FG_ARCH Linux-x86-64)
  endif()
  # The $<1:...> generator expression stops multi-config generators (Visual Studio)
  # from appending a Release/ or Debug/ subfolder, which hosts would not find.
  set_target_properties(FacetGlass PROPERTIES
    LIBRARY_OUTPUT_DIRECTORY $<1:${CMAKE_BINARY_DIR}/FacetGlass.ofx.bundle/Contents/${FG_ARCH}>)
  if(MSVC)
    target_compile_definitions(FacetGlass PRIVATE NOMINMAX _CRT_SECURE_NO_WARNINGS)
    target_compile_options(FacetGlass PRIVATE /W3 /fp:precise)
    # Static CRT: the plugin must load in any host without a VC++ redistributable.
    set_target_properties(FacetGlass PROPERTIES MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  endif()
endif()
